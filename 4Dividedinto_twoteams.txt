def imprimirMatriz(titulo, matriz):

    ancho = 1
    for fila in matriz:
        for valor in fila:
            if len(str(valor)) > ancho:
                ancho = len(str(valor))

    print(titulo)
    for fila in matriz:
        textos = []
        for valor in fila:
            textos.append(str(valor).rjust(ancho))
        print(" ".join(textos))
    print()


def matConsecutiva(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            fila.append(f * n + c + 1)
        matriz.append(fila)
    return matriz


def matSubeBaja(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if f % 2 == 0:
                fila.append(c + 1)
            else:
                fila.append(n - c)
        matriz.append(fila)
    return matriz


# 3. Unos en la diagonal principal, ceros en el resto
def matDiagonalPrincipal(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if f == c:
                fila.append(1)
            else:
                fila.append(0)
        matriz.append(fila)
    return matriz



# 4. Unos en la diagonal secundaria, ceros en el resto
def matDiagonalSecundaria(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if c == n - 1 - f:
                fila.append(1)
            else:
                fila.append(0)
        matriz.append(fila)
    return matriz


# 5. Unos en la diagonal y debajo de ella (triángulo inferior)
def matTrianguloInferior(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if c <= f:
                fila.append(1)
            else:
                fila.append(0)
        matriz.append(fila)
    return matriz


# 6. Unos en la diagonal y encima de ella (triángulo superior)
def matTrianguloSuperior(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if c >= f:
                fila.append(1)
            else:
                fila.append(0)
        matriz.append(fila)
    return matriz


# 7. Números de 1 a n*n, columna por columna
def matPorColumnas(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            fila.append(c * n + f + 1)
        matriz.append(fila)
    return matriz



def matLetras(n):
    matriz = []
    for f in range(n):
        fila = []
        for c in range(n):
            if c <= f:
                if c % 2 == 0:
                    fila.append("A")
                else:
                    fila.append("B")
            else:
                fila.append("")
        matriz.append(fila)
    return matriz

print()
imprimirMatriz("Matriz 1 (consecutiva por filas):", matConsecutiva(n))
imprimirMatriz("Matriz 2 (filas que suben y bajan):", matSubeBaja(n))
imprimirMatriz("Matriz 3 (diagonal principal):", matDiagonalPrincipal(n))
imprimirMatriz("Matriz 4 (diagonal secundaria):", matDiagonalSecundaria(n))
imprimirMatriz("Matriz 5 (triángulo inferior):", matTrianguloInferior(n))
imprimirMatriz("Matriz 6 (triángulo superior):", matTrianguloSuperior(n))
